# Alert

Inline status message. Uses `role="alert"` for danger/warning variants so assistive tech announces it immediately; `role="status"` otherwise.

<LiveDemo code={`<Alert variant="warning" title="Payment method expiring">Update your card before the end of the month.</Alert>`} />

## Variants

<LiveDemo code={`<div className="ui-flex ui-w-full ui-flex-col ui-gap-3">

    <Alert variant="success" title="Saved">Your changes were saved successfully.</Alert>
    <Alert variant="danger" title="Something went wrong">The request could not be completed.</Alert>
  </div>`} minHeight={260} />

import { Alert } from "@labanaat/ui/alert";
